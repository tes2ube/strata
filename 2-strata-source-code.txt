`export function createSettings(storage = globalThis.localStorage, storageKey = 'strata.settings.v1') {`

`let overrides = {};`

`const listeners = new Set();`

`try { overrides = JSON.parse(storage?.getItem(storageKey) || '{}'); } catch { overrides = {}; }`

`function coerce(key, v) {`

`const m = META[key];`

`if (!m) return v;`

`if (m.type === 'number') { let n = Number(v); if (!Number.isFinite(n)) n = m.default; if (m.min != null) n = Math.max(m.min, n); if (m.max != null) n = Math.min(m.max, n); return n; }`

`if (m.type === 'boolean') return v === true || v === 'true';`

`return String(v);`

`}`

`const api = {`

`get(key) { if (!(key in DEFAULTS)) throw new Error('Unknown setting: ' + key); return key in overrides ? overrides[key] : DEFAULTS[key]; },`

`set(key, v) { overrides[key] = coerce(key, v); if (overrides[key] === DEFAULTS[key]) delete overrides[key]; persist(); },`

`reset(key) { if (key) delete overrides[key]; else overrides = {}; persist(); },`

`isModified(key) { return key in overrides; },`

`all() { return { ...DEFAULTS, ...overrides }; },`

`exportJSON() { return JSON.stringify(overrides, null, 2); },`

`// Hex lattice. Works rest on lattice points ("snaps"). Rows are horizontal; neighbors sit at 0, 60, 120... degrees.`

`const S3 = Math.sqrt(3) / 2;`

`export function makeLattice(spacing, origin = { x: 0, y: 0 }) {`

`const dx = spacing, dy = spacing * S3;`

`const pointOf = (q, r) => ({ x: origin.x + dx * (q + r / 2), y: origin.y + dy * r });`

`const toAxial = (x, y) => {`

`const r = (y - origin.y) / dy, q = (x - origin.x) / dx - r / 2;`

`let cx = Math.round(q), cz = Math.round(r), cy = Math.round(-q - r);`

`const ex = Math.abs(cx - q), ez = Math.abs(cz - r), ey = Math.abs(cy + q + r);`

`if (ex > ey && ex > ez) cx = -cy - cz; else if (ez > ey) cz = -cx - cy;`

`return { q: cx, r: cz };`

`};`

`const key = (q, r) => q + ',' + r;`

`const dist = (a, b) => (Math.abs(a.q - b.q) + Math.abs(a.q + a.r - b.q - b.r) + Math.abs(a.r - b.r)) / 2;`

`// nearest unoccupied lattice point to (x,y), searching rings outward`

`function nearestFree(x, y, occupied, blocked = () => false) {`

`const c = toAxial(x, y);`

`let best = null;`

`for (let k = 0; k < 40; k++) {`

`for (const h of ring(c, k)) {`

`if (occupied.has(key(h.q, h.r))) continue;`

`const p = pointOf(h.q, h.r);`

`if (blocked(p)) continue;`

`const d = Math.hypot(p.x - x, p.y - y);`

`if (!best || d < best.d) best = { ...h, d, p };`

`}`

`if (best && best.d <= (k + 0.5) * dy * 0.9) break; // nothing farther ring can beat it`

`}`

`return best;`

`}`

`function ring(c, k) {`

`if (k === 0) return [c];`

`const dirs = [[1, 0], [0, 1], [-1, 1], [-1, 0], [0, -1], [1, -1]];`

`const out = [];`

`let q = c.q + dirs[4][0] * k, r = c.r + dirs[4][1] * k;`

`for (let s = 0; s < 6; s++) for (let i = 0; i < k; i++) { out.push({ q, r }); q += dirs[s][0]; r += dirs[s][1]; }`

`return out;`

`}`

`return { dx, dy, origin, pointOf, toAxial, key, dist, nearestFree, ring };`

`}`

`// Layout: fixed genre anchors + force simulation to find rough positions, then snap each work to the nearest free hex point.`

`// Relatedness sets spring length: weakly related works end up a few hex points apart.`

`import { FAMILY_IDS } from '../settings/definitions.js';`

`import { makeLattice } from './hex.js';`

`function rng(seed) { let a = seed >>> 0; return () => { a |= 0; a = (a + 0x6D2B79F5) | 0; let t = Math.imul(a ^ (a >>> 15), 1 | a); t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t; return ((t ^ (t >>> 14)) >>> 0) / 4294967296; }; }`

`export function anchorsFor(cfg) {`

`const R = cfg.get('genres.ringRadius'); const n = FAMILY_IDS.length;`

`return FAMILY_IDS.map((_, i) => { const a = -Math.PI / 2 + i * 2 * Math.PI / n; return { x: R * Math.cos(a), y: R * Math.sin(a) }; });`

`}`

`export function layout({ works, edges, cfg, prev = null }) {`

`const lattice = makeLattice(cfg.get('layout.spacing'));`

`const sp = cfg.get('layout.spacing'); const rand = rng(cfg.get('layout.seed'));`

`const anchors = anchorsFor(cfg);`

`const idx = new Map(works.map((w, i) => [w.id, i]));`

`const target = works.map((w) => { let x = 0, y = 0, s = 0; for (const [f, wt] of Object.entries(w.fam)) { x += anchors[f].x * wt; y += anchors[f].y * wt; s += wt; } return { x: x / s, y: y / s }; });`

`const P = works.map((w, i) => prev && prev[w.id] ? { ...prev[w.id] } : { x: target[i].x + (rand() - 0.5) * sp, y: target[i].y + (rand() - 0.5) * sp });`

`const mass = works.map((w) => w.heard ? 1 + cfg.get('layout.massScale') * ((w.rating || 6) - 5) : 1);`

`const iters = cfg.get('layout.iterations'), cool = cfg.get('layout.cooling');`

`const range = cfg.get('layout.repulsionRange') * sp, rep = cfg.get('layout.repulsion'), well = cfg.get('genres.wellStrength');`

`const k = cfg.get('layout.linkStrength'), base = cfg.get('layout.linkRest') * sp, stretch = cfg.get('layout.relatednessStretch');`

`let alpha = 1;`

`for (let it = 0; it < iters; it++) {`

`const F = works.map(() => ({ x: 0, y: 0 }));`

`for (let i = 0; i < works.length; i++) {`

`F[i].x += (target[i].x - P[i].x) * well; F[i].y += (target[i].y - P[i].y) * well;`

`for (let j = i + 1; j < works.length; j++) {`

`const dx = P[i].x - P[j].x, dy = P[i].y - P[j].y, d = Math.hypot(dx, dy) || 0.01;`

`if (d < range) { const f = rep * (range - d) / range * sp * 0.1; const ux = dx / d, uy = dy / d; F[i].x += ux * f * Math.sqrt(mass[j]); F[i].y += uy * f * Math.sqrt(mass[j]); F[j].x -= ux * f * Math.sqrt(mass[i]); F[j].y -= uy * f * Math.sqrt(mass[i]); }`

`}`

`}`

`for (const e of edges) {`

`const i = idx.get(e.a), j = idx.get(e.b); if (i == null || j == null) continue;`

`const rest = base * (1 + stretch * (1 - (e.sim ?? 0.7)));`

`const dx = P[j].x - P[i].x, dy = P[j].y - P[i].y, d = Math.hypot(dx, dy) || 0.01, f = k * (d - rest);`

`F[i].x += dx / d * f; F[i].y += dy / d * f; F[j].x -= dx / d * f; F[j].y -= dy / d * f;`

`}`

`for (let i = 0; i < works.length; i++) { if (works[i].pinned) continue; P[i].x += Math.max(-sp * .3, Math.min(sp * .3, F[i].x)) * alpha; P[i].y += Math.max(-sp * .3, Math.min(sp * .3, F[i].y)) * alpha; }`

`alpha *= cool;`

`}`

`// snap: heavy/heard first, then by closeness to center`

`const hub = cfg.get('layout.hubRadius');`

`const order = works.map((w, i) => i).sort((a, b) => (works[b].heard - works[a].heard) || (mass[b] - mass[a]) || (Math.hypot(P[a].x, P[a].y) - Math.hypot(P[b].x, P[b].y)));`

`const occ = new Set(); const hex = {}, pos = {};`

`for (const i of order) {`

`const r = lattice.nearestFree(P[i].x, P[i].y, occ, (pt) => Math.hypot(pt.x, pt.y) < hub);`

`occ.add(lattice.key(r.q, r.r)); hex[works[i].id] = { q: r.q, r: r.r }; pos[works[i].id] = { x: r.p.x, y: r.p.y };`

`}`

`return { pos, hex, lattice, anchors, raw: Object.fromEntries(works.map((w, i) => [w.id, P[i]])) };`

`}`

`// Connection routing on the hex lattice.`

`// Straight runs travel along the three hex axes (0, 60, 120 degrees) in "lanes" half a row away from the tiles,`

`// so they never cross a work. A route is: short perpendicular stub out of a tile -> lane -> (turn) -> lane -> stub into the other tile.`

`// Lines that share a work (fan-out or fan-in) are bundled: they use the same stub and the same first lane, and branch off where they must.`

`const S3 = Math.sqrt(3) / 2;`

`export const AXES = [{ x: 1, y: 0 }, { x: 0.5, y: S3 }, { x: -0.5, y: S3 }];`

`const normalOf = (a) => ({ x: -a.y, y: a.x });`

`const add = (p, q, k = 1) => ({ x: p.x + q.x * k, y: p.y + q.y * k });`

`const sub = (p, q) => ({ x: p.x - q.x, y: p.y - q.y });`

`const dot = (p, q) => p.x * q.x + p.y * q.y;`

`const cross = (p, q) => p.x * q.y - p.y * q.x;`

`const len = (p) => Math.hypot(p.x, p.y);`

`const near = (p, q) => Math.hypot(p.x - q.x, p.y - q.y) < 0.01;`

`// Lane legs (everything between the first and last stub) as {ax, lane, t0, t1}`

`function laneLegs(points) {`

`const out = [];`

`for (let k = 1; k < points.length - 2; k++) {`

`const a = points[k], b = points[k + 1], d = sub(b, a), l = len(d); if (l < 1) continue;`

`const u = { x: d.x / l, y: d.y / l };`

`const ax = AXES.findIndex((A) => Math.abs(cross(A, u)) < 0.01); if (ax < 0) continue;`

`const A = AXES[ax], t0 = dot(a, A), t1 = dot(b, A);`

`out.push({ ax, lane: cross(A, a), t0: Math.min(t0, t1), t1: Math.max(t0, t1) });`

`}`

`return out;`

`}`

`function overlapLength(points, placed, key) {`

`let total = 0;`

`for (const L of laneLegs(points)) for (const q of placed) {`

`if (q.ax !== L.ax || Math.abs(q.lane - L.lane) > 1 || (key && q.key === key)) continue;`

`total += Math.max(0, Math.min(L.t1, q.t1) - Math.max(L.t0, q.t0));`

`}`

`return total;`

`}`

`function combos() { const o = []; for (let ax = 0; ax < 3; ax++) for (const side of [1, -1]) o.push({ ax, side }); return o; }`

`const COMBOS = combos();`

`function endpoint(c, half, combo, h) {`

`const n = normalOf(AXES[combo.ax]);`

`return { port: add(c, n, combo.side * (half + 2)), gate: add(c, n, combo.side * h), ax: combo.ax, side: combo.side };`

`}`

`// Route from S to T with fixed combos. Returns {points, cost} or null.`

`function routeFixed(S, T, es, et, p) {`

`const d1 = AXES[es.ax], d2 = AXES[et.ax], G1 = es.gate, G2 = et.gate;`

`const toT = sub(T.c, S.c);`

`let pts, legs = 0, back = 0, turns;`

`if (es.ax === et.ax) {`

`if (Math.abs(cross(sub(G2, G1), d1)) > 1) return null; // different parallel lanes`

`pts = [es.port, G1, G2, et.port]; legs = len(sub(G2, G1)); turns = 2;`

`if (dot(sub(G2, G1), toT) < 0) back = legs;`

`} else {`

`const cr = cross(d1, d2), w = sub(G2, G1);`

`const t = cross(w, d2) / cr, u = cross(w, d1) / cr;`

`const X = add(G1, d1, t);`

`pts = [es.port, G1, X, G2, et.port]; legs = Math.abs(t) + Math.abs(u); turns = 3;`

`if (dot({ x: d1.x * t, y: d1.y * t }, toT) < 0) back += Math.abs(t);`

`if (dot({ x: -d2.x * u, y: -d2.y * u }, toT) < 0) back += Math.abs(u);`

`}`

`const clean = [pts[0]];`

`for (let i = 1; i < pts.length; i++) if (!near(pts[i], clean[clean.length - 1])) clean.push(pts[i]);`

`const stub = len(sub(es.gate, es.port)) + len(sub(et.gate, et.port));`

`const ov = p.placed.length ? overlapLength(clean, p.placed, p.key) : 0;`

`return { points: clean, cost: legs + stub + p.turn * turns + p.back * back + p.overlap * ov };`

`}`

`export function ratingFactor(rating, cfg) {`

`const floor = cfg.get('tiers.ratingFloor'), ceil = cfg.get('tiers.ratingCeil'), top = cfg.get('tiers.ratingTopFactor');`

`if (rating <= floor) return 0;`

`if (rating >= ceil) return top;`

`return top * (rating - floor) / (ceil - floor);`

`}`

`export function computeMastery(works, cfg) {`

`const acc = {}; // fam -> tier -> {sum, cnt}`

`for (const f of FAMILY_IDS) acc[f] = TIER_IDS.map(() => ({ sum: 0, cnt: 0 }));`

`for (const w of works) {`

`if (!w.heard) continue;`

`let bm = 1;`

``for (const b of w.badges || []) if (cfg.get(`badges.${b}.enabled`)) bm *= cfg.get(`badges.${b}.masteryMult`);``

``const val = cfg.get(`tiers.${TIER_IDS[w.tier]}.masteryValue`) * ratingFactor(w.rating, cfg) * bm;``

`for (const [fam, wt] of Object.entries(w.fam)) {`

`const slot = acc[FAMILY_IDS[fam]][w.tier];`

`slot.sum += val * wt; slot.cnt += wt;`

`}`

`}`

`const exp = cfg.get('tiers.diminishing');`

`const mastery = {}, counts = {};`

`for (const f of FAMILY_IDS) {`

`let m = 0; counts[f] = [];`

`acc[f].forEach((s, t) => { counts[f][t] = s.cnt; if (s.cnt > 0) m += (s.sum / s.cnt) * Math.pow(s.cnt, exp); });`

`mastery[f] = m;`

`}`

`return { mastery, counts };`

`}`

`// Vertical rating bar: continuous position -> light/decent/strong steps, with border zones between steps.`

`// f is 0 at the bottom of the bar and 1 at the top. Each integer owns an equal slice; each slice is split in thirds.`

`export function ratingFromFraction(f, cfg) {`

`const lo = cfg.get('rating.min'), hi = cfg.get('rating.max');`

`const step = cfg.get('rating.subStep'), band = cfg.get('rating.borderBand');`

`const slices = hi - lo + 1;`

`const v = Math.min(slices - 1e-9, Math.max(0, f * slices)); // 0 .. slices`

`const n = lo + Math.floor(v);`

`const within = v - Math.floor(v); // 0..1 inside the integer slice`

`const thirds = within * 3; // 0..3`

`const idx = Math.min(2, Math.floor(thirds));`

`const fracInThird = thirds - Math.floor(thirds);`

`const names = ['light', 'decent', 'strong'];`

`const val = (nn, i) => nn + (i - 1) * step;`

`const nearLow = fracInThird < band / 2, nearHigh = fracInThird > 1 - band / 2;`

`let kind = 'step', label, value, parts;`

`if (nearLow && !(idx === 0 && n === lo)) {`

`// border with the previous step`

`const pi = idx === 0 ? 2 : idx - 1, pn = idx === 0 ? n - 1 : n;`

`kind = 'border';`

`value = (val(pn, pi) + val(n, idx)) / 2;`

``label = idx === 0 ? `strong ${pn} / light ${n}` : `${names[pi]}/${names[idx]} ${n}`;``

`parts = [[pn, names[pi]], [n, names[idx]]];`

`} else if (nearHigh && !(idx === 2 && n === hi)) {`

`const ni = idx === 2 ? 0 : idx + 1, nn = idx === 2 ? n + 1 : n;`

`kind = 'border';`

`value = (val(n, idx) + val(nn, ni)) / 2;`

``label = idx === 2 ? `strong ${n} / light ${nn}` : `${names[idx]}/${names[ni]} ${n}`;``

`parts = [[n, names[idx]], [nn, names[ni]]];`

`} else {`

`value = val(n, idx);`

``label = `${names[idx]} ${n}`;``

`parts = [[n, names[idx]]];`

`}`

`return { n, kind, label, value: Math.round(value * 1000) / 1000, parts };`

`}`

`const wait = Math.max(0, lastMB + cfg.get('data.musicBrainzDelayMs') - Date.now());`

`}`
